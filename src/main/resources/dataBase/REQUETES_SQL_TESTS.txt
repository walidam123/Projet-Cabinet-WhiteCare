-- 11. VÉRIFICATION DES DONNÉES INSÉRÉES
-- ============================================================================

-- Afficher le nombre d'enregistrements par table
SELECT 'Cabinet' AS TableName, COUNT(*) AS Nombre FROM cabinet_medicale
UNION ALL
SELECT 'Utilisateurs', COUNT(*) FROM utilisateur
UNION ALL
SELECT 'Staff', COUNT(*) FROM staff
UNION ALL
SELECT 'Médecins', COUNT(*) FROM medecin
UNION ALL
SELECT 'Secrétaires', COUNT(*) FROM secretaire
UNION ALL
SELECT 'Charges', COUNT(*) FROM charges
UNION ALL
SELECT 'Revenus', COUNT(*) FROM revenues;

-- Afficher le total des charges
SELECT 
    'Total Charges' AS Type,
    COUNT(*) AS Nombre,
    SUM(montant) AS Total_MAD,
    MIN(date) AS Date_Min,
    MAX(date) AS Date_Max
FROM charges
WHERE cabinet_medicale_id = (SELECT id FROM cabinet_medicale WHERE nom = 'Cabinet WhiteCare' LIMIT 1);

-- Afficher le total des revenus
SELECT 
    'Total Revenus' AS Type,
    COUNT(*) AS Nombre,
    SUM(montant) AS Total_MAD,
    AVG(montant) AS Moyenne_MAD,
    MIN(date) AS Date_Min,
    MAX(date) AS Date_Max
FROM revenues
WHERE cabinet_medicale_id = (SELECT id FROM cabinet_medicale WHERE nom = 'Cabinet WhiteCare' LIMIT 1);

-- Afficher le solde (Revenus - Charges)
SELECT 
    (SELECT COALESCE(SUM(montant), 0) FROM revenues WHERE cabinet_medicale_id = (SELECT id FROM cabinet_medicale WHERE nom = 'Cabinet WhiteCare' LIMIT 1)) -
    (SELECT COALESCE(SUM(montant), 0) FROM charges WHERE cabinet_medicale_id = (SELECT id FROM cabinet_medicale WHERE nom = 'Cabinet WhiteCare' LIMIT 1)) AS Solde_MAD;
